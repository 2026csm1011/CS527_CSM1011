% faq5.txt - FIR Filter Response
% Initialize Pointers
x10 = 0
x1 = [x10]
x2 = 4
x3 = 16
x4 = 0

firloop
x11 = x1 - 0
BEQ endfir

% Apply weights W0=5, W1=2
x5 = [x2]
x6 = x5 * 5
x7 = x4 * 2
x8 = x6 + x7

% Store Output
[x3] = x8
% Trick to copy x5 into x4 for the next loop!
x4 = x5 + 0
x2 = x2 + 4
x3 = x3 + 4
x1 = x1 - 1
BAL firloop

endfir
x10 = 0
